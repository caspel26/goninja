{{- /* Renders a star-count chip linking to the repository. The count is fetched at build time, not from the browser, so a page view costs no extra network request and the number can't rate-limit a visitor. It goes stale between builds, which is fine for a star count. Any failure — offline, rate-limited, unexpected payload — renders nothing rather than failing the build. `GITHUB_TOKEN` is used when present, since unauthenticated GitHub API calls are rate-limited per IP and CI runners share addresses. */ -}} {{- with .Site.Params.repo -}} {{- $url := printf "https://api.github.com/repos/%s" . -}} {{- $headers := dict "Accept" "application/vnd.github+json" -}} {{- with getenv "GITHUB_TOKEN" -}} {{- $headers = merge $headers (dict "Authorization" (printf "Bearer %s" .)) -}} {{- end -}} {{- $stars := 0 -}} {{- with try (resources.GetRemote $url (dict "headers" $headers)) -}} {{- if not .Err -}} {{- with .Value -}} {{- with try (transform.Unmarshal .Content) -}} {{- if not .Err -}} {{- $stars = (int (index .Value "stargazers_count" | default 0)) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if gt $stars 0 -}} {{- $label := printf "%d" $stars -}} {{- if ge $stars 1000 -}} {{- $label = printf "%.1fk" (div (float $stars) 1000.0) -}} {{- end -}} {{ $label }} {{- end -}} {{- end -}}
